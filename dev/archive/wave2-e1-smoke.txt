ENGINE SMOKE — WC-E1-ENGINE
------------------------------------------------------------------------
Goal: source-inspection that the script + migration both exist and are
structurally sound (no DB call required — runs in CI without env).

Suggested test (inside scripts/smoke_test.py):

  @test("WC-E1-ENGINE — Monte Carlo script + migration structure")
  def test_wc_e1_engine():
      from pathlib import Path
      root = Path(__file__).resolve().parent.parent

      # 1. Script exists and is importable as a module.
      script = root / "scripts" / "wc_monte_carlo.py"
      assert script.exists(), "scripts/wc_monte_carlo.py missing"
      src = script.read_text()
      assert "def run_simulations" in src, "run_simulations missing"
      assert "def run_wc_monte_carlo" in src, "scheduler entry-point missing"
      assert "wc_monte_carlo_results" in src, "table reference missing"
      assert "predict_1x2_from_elo" in src, "ELO predictor not used"

      # 2. Migration exists with the right table + RLS public-read.
      mig = root / "supabase" / "migrations" / "179_wc_monte_carlo.sql"
      assert mig.exists(), "migration 179_wc_monte_carlo.sql missing"
      msrc = mig.read_text()
      assert "CREATE TABLE IF NOT EXISTS wc_monte_carlo_results" in msrc
      assert "p_advance" in msrc and "p_winner" in msrc
      assert "ENABLE ROW LEVEL SECURITY" in msrc
      assert 'Public read' in msrc

      # 3. Scheduler wiring — single new cron entry at 06:30 UTC.
      sched = (root / "workers" / "scheduler.py").read_text()
      assert "job_wc_monte_carlo" in sched, "scheduler job fn missing"
      assert "wc_monte_carlo" in sched and "hour=6, minute=30" in sched

Notes:
  - Source-inspection only — does not require DB or network.
  - Mirrors the WC-A3 / WC-AI-PREVIEW smoke pattern (see smoke_test.py
    around the wc_market_consensus / wc_match_previews entries).
  - Verified locally: dry-run `python3 scripts/wc_monte_carlo.py --dry-run
    --n-sims 1000` completes in ~0.2s and prints the top-10 table.
